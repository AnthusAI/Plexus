To access the GraphQL Proxy:

  # Port forward to local machine
  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ .Release.Name }}-graphql-proxy 8000:8000

  # Then access at: http://localhost:8000/graphql
{{- end }}

{{- if .Values.postgresql.enabled }}

To connect to PostgreSQL:

  # Get the password
  export POSTGRES_PASSWORD=$(kubectl get secret -n {{ .Release.Namespace }} {{ .Release.Name }}-postgresql -o jsonpath="{.data.password}" | base64 -d)

  # Port forward
  kubectl port-forward -n {{ .Release.Namespace }} svc/{{ .Release.Name }}-postgresql 5432:5432

  # Connect
  psql -h localhost -U {{ .Values.postgresql.auth.username }} -d {{ .Values.postgresql.auth.database }}
{{- end }}
